-- ============================================================================ -- 04_sales.sql - CAF-based sales save/edit + cash-sale invoice numbers. Safe to re-run. -- One CAF (account_no) = one sale; all its items belong to it. -- Cash sale = invoice number INV-000001, INV-000002 ... stored in sales.account_no -- ============================================================================ create sequence if not exists public.cash_invoice_seq start 1; create or replace function public.next_invoice_no() returns text language plpgsql security definer set search_path = public as $$ begin if coalesce(public.app_role(), '') not in ('admin','accountant') then raise exception 'You do not have permission'; end if; return 'INV-' || lpad(nextval('public.cash_invoice_seq')::text, 6, '0'); end $$; drop function if exists public.save_sale(jsonb, jsonb); create or replace function public.save_sale(p_sale jsonb, p_items jsonb, p_replace boolean default false) returns jsonb language plpgsql security definer set search_path = public as $$ declare it jsonb; v_acc text := btrim(p_sale->>'account_no'); v_pid text; v_qty numeric; v_bal numeric; n int := 0; begin if coalesce(public.app_role(), '') not in ('admin','accountant') then raise exception 'You do not have permission to save sales'; end if; if v_acc is null or v_acc = '' then raise exception 'CAF / invoice number is required'; end if; if p_items is null or jsonb_typeof(p_items) <> 'array' or jsonb_array_length(p_items) = 0 then raise exception 'Sale has no items'; end if; if exists (select 1 from sales where btrim(account_no) = v_acc) then if not p_replace then raise exception 'Sale against % is already saved', v_acc; end if; delete from sales where btrim(account_no) = v_acc; -- edit = replace the whole sale delete from stock_ledger where reference_type = 'SALE' and btrim(reference_id) = v_acc; -- and give its stock back end if; for it in select * from jsonb_array_elements(p_items) loop v_pid := nullif(it->>'product_id',''); v_qty := coalesce((it->>'qty')::numeric, 1); insert into sales(product_id, date, approved_by, account_no, account_title, company_name, product_category, model, color, qty, sales_rate, sales_amt, plan_months, advance, installment_per_month, processing_fee, total_amt, next_inst_month, remarks) values (v_pid, (p_sale->>'date')::date, p_sale->>'approved_by', v_acc, p_sale->>'account_title', it->>'company_name', it->>'product_category', it->>'model', it->>'color', v_qty, coalesce((it->>'sales_rate')::numeric,0), coalesce((it->>'sales_amt')::numeric,0), coalesce((it->>'plan_months')::numeric,0), coalesce((it->>'advance')::numeric,0), coalesce((it->>'installment_per_month')::numeric,0), coalesce((it->>'processing_fee')::numeric,0), coalesce((it->>'sales_amt')::numeric,0) + coalesce((it->>'processing_fee')::numeric,0), p_sale->>'next_inst_month', it->>'remarks'); if v_pid is not null then select coalesce(sum(coalesce(qty_in,0) - coalesce(qty_out,0)), 0) into v_bal from stock_ledger where product_id = v_pid; insert into stock_ledger(transaction_id, date, product_id, company_name, product_category, model, color, qty_in, qty_out, balance, reference_type, reference_id, warehouse, remarks) values ('OUT-' || upper(substr(md5(random()::text || clock_timestamp()::text), 1, 9)), (p_sale->>'date')::date, v_pid, it->>'company_name', it->>'product_category', it->>'model', it->>'color', 0, v_qty, v_bal - v_qty, 'SALE', v_acc, 'MAIN WAREHOUSE', 'Sale Acc ' || v_acc); end if; n := n + 1; end loop; return jsonb_build_object('success', true, 'items', n, 'account_no', v_acc); end $$; revoke execute on function public.save_sale(jsonb,jsonb,boolean), public.next_invoice_no() from public, anon; grant execute on function public.save_sale(jsonb,jsonb,boolean), public.next_invoice_no() to authenticated;
